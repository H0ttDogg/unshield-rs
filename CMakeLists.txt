cmake_minimum_required(VERSION 3.21)
project(unshield VERSION 1.6.2 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)
option(BUILD_STATIC "Install the Rust static library instead of the shared library" OFF)
option(BUILD_CLI "Build and install the Rust command-line tool" ON)
option(TEST_UPSTREAM_CLI "Compile the unchanged pinned upstream C CLI against Rust" OFF)
option(TEST_REAL_INSTALLERS "Require downloaded real installers in C ABI tests" OFF)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(RUST_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo")
set(RUST_ENV "CARGO_TARGET_DIR=${RUST_TARGET_DIR}")
if(UNIX AND NOT APPLE AND NOT BUILD_STATIC)
    # The upstream ABI uses major version 1. This permits replacement for already-linked callers.
    list(APPEND RUST_ENV "RUSTFLAGS=-C link-arg=-Wl,-soname,libunshield.so.1")
endif()
if(APPLE AND NOT BUILD_STATIC)
    list(APPEND RUST_ENV "RUSTFLAGS=-C link-arg=-Wl,-install_name,@rpath/libunshield.1.dylib")
endif()
if(WIN32)
    set(RUST_SHARED "${RUST_TARGET_DIR}/release/unshield.dll")
    set(RUST_IMPORT "${RUST_TARGET_DIR}/release/unshield.dll.lib")
    set(RUST_STATIC "${RUST_TARGET_DIR}/release/unshield.lib")
    set(RUST_EXE "${RUST_TARGET_DIR}/release/unshield-rs.exe")
elseif(APPLE)
    set(RUST_SHARED "${RUST_TARGET_DIR}/release/libunshield.dylib")
    set(RUST_STATIC "${RUST_TARGET_DIR}/release/libunshield.a")
    set(RUST_EXE "${RUST_TARGET_DIR}/release/unshield-rs")
else()
    set(RUST_SHARED "${RUST_TARGET_DIR}/release/libunshield.so")
    set(RUST_STATIC "${RUST_TARGET_DIR}/release/libunshield.a")
    set(RUST_EXE "${RUST_TARGET_DIR}/release/unshield-rs")
endif()
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.rs")
add_custom_command(OUTPUT "${RUST_SHARED}" "${RUST_STATIC}"
    COMMAND "${CMAKE_COMMAND}" -E env ${RUST_ENV} "${CARGO_EXECUTABLE}" build --locked --release --lib
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock VERBATIM)
add_custom_target(rust_library ALL DEPENDS "${RUST_SHARED}" "${RUST_STATIC}")
if(UNIX AND NOT BUILD_STATIC)
    if(APPLE)
        set(RUST_MAJOR_ALIAS "${RUST_TARGET_DIR}/release/libunshield.1.dylib")
        set(RUST_SHARED_NAME libunshield.dylib)
    else()
        set(RUST_MAJOR_ALIAS "${RUST_TARGET_DIR}/release/libunshield.so.1")
        set(RUST_SHARED_NAME libunshield.so)
    endif()
    add_custom_command(OUTPUT "${RUST_SHARED}" APPEND
        COMMAND "${CMAKE_COMMAND}" -E create_symlink "${RUST_SHARED_NAME}" "${RUST_MAJOR_ALIAS}")
endif()
if(BUILD_STATIC)
    add_library(unshield_rust_artifact STATIC IMPORTED)
    set_target_properties(unshield_rust_artifact PROPERTIES IMPORTED_LOCATION "${RUST_STATIC}")
else()
    add_library(unshield_rust_artifact SHARED IMPORTED)
    set_target_properties(unshield_rust_artifact PROPERTIES IMPORTED_LOCATION "${RUST_SHARED}")
    if(WIN32)
        set_target_properties(unshield_rust_artifact PROPERTIES IMPORTED_IMPLIB "${RUST_IMPORT}")
    elseif(APPLE)
        set_target_properties(unshield_rust_artifact PROPERTIES IMPORTED_SONAME "@rpath/libunshield.1.dylib")
    else()
        set_target_properties(unshield_rust_artifact PROPERTIES IMPORTED_SONAME "libunshield.so.1")
    endif()
endif()
add_dependencies(unshield_rust_artifact rust_library)
add_library(libunshield INTERFACE)
add_dependencies(libunshield rust_library)
target_include_directories(libunshield INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(BUILD_STATIC)
    target_link_libraries(libunshield INTERFACE $<BUILD_INTERFACE:unshield_rust_artifact> $<INSTALL_INTERFACE:unshield>)
    if(WIN32)
        target_link_libraries(libunshield INTERFACE kernel32 ntdll userenv ws2_32 dbghelp bcrypt advapi32)
    elseif(APPLE)
        target_link_libraries(libunshield INTERFACE "-framework Security" "-framework SystemConfiguration" resolv)
    else()
        target_link_libraries(libunshield INTERFACE dl pthread m rt util)
    endif()
elseif(WIN32)
    target_compile_definitions(libunshield INTERFACE UNSHIELD_DYNAMIC_LIBRARY)
    target_link_libraries(libunshield INTERFACE $<BUILD_INTERFACE:unshield_rust_artifact> $<INSTALL_INTERFACE:unshield.dll.lib>)
else()
    target_link_libraries(libunshield INTERFACE $<BUILD_INTERFACE:unshield_rust_artifact> $<INSTALL_INTERFACE:unshield>)
endif()
add_library(unshield::libunshield ALIAS libunshield)
if(BUILD_CLI)
    add_custom_command(OUTPUT "${RUST_EXE}"
        COMMAND "${CMAKE_COMMAND}" -E env ${RUST_ENV} "${CARGO_EXECUTABLE}" build --locked --release --bin unshield-rs
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        DEPENDS rust_library ${RUST_SOURCES} Cargo.toml Cargo.lock VERBATIM)
    add_custom_target(rust_cli ALL DEPENDS "${RUST_EXE}")
    install(PROGRAMS "${RUST_EXE}" DESTINATION "${CMAKE_INSTALL_BINDIR}" RENAME "unshield${CMAKE_EXECUTABLE_SUFFIX}")
endif()
if(BUILD_STATIC)
    install(FILES "${RUST_STATIC}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
elseif(WIN32)
    install(FILES "${RUST_SHARED}" DESTINATION "${CMAKE_INSTALL_BINDIR}")
    install(FILES "${RUST_IMPORT}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
elseif(APPLE)
    install(FILES "${RUST_SHARED}" DESTINATION "${CMAKE_INSTALL_LIBDIR}" RENAME "libunshield.${PROJECT_VERSION}.dylib")
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink libunshield.${PROJECT_VERSION}.dylib \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libunshield.1.dylib\")")
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink libunshield.1.dylib \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libunshield.dylib\")")
else()
    install(FILES "${RUST_SHARED}" DESTINATION "${CMAKE_INSTALL_LIBDIR}" RENAME "libunshield.so.${PROJECT_VERSION}")
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink libunshield.so.${PROJECT_VERSION} \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libunshield.so.1\")")
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E create_symlink libunshield.so.1 \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/libunshield.so\")")
endif()
install(FILES include/libunshield.h DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
file(RELATIVE_PATH PC_PREFIX_RELATIVE "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
set(PC_PREFIX "\${pcfiledir}/${PC_PREFIX_RELATIVE}")
if(IS_ABSOLUTE "${PC_PREFIX_RELATIVE}")
    set(PC_PREFIX "${CMAKE_INSTALL_PREFIX}")
endif()
foreach(PC_DIR LIBDIR INCLUDEDIR)
    if(IS_ABSOLUTE "${CMAKE_INSTALL_${PC_DIR}}")
        set(PC_${PC_DIR} "${CMAKE_INSTALL_${PC_DIR}}")
    else()
        set(PC_${PC_DIR} "\${prefix}/${CMAKE_INSTALL_${PC_DIR}}")
    endif()
endforeach()
set(PC_LIBRARY "-lunshield")
if(WIN32)
    set(PC_PRIVATE_LIBS "-lkernel32 -lntdll -luserenv -lws2_32 -ldbghelp -lbcrypt -ladvapi32")
    if(NOT BUILD_STATIC)
        set(PC_LIBRARY "\"\${libdir}/unshield.dll.lib\"")
    endif()
elseif(APPLE)
    set(PC_PRIVATE_LIBS "-framework Security -framework SystemConfiguration -lresolv")
else()
    set(PC_PRIVATE_LIBS "-ldl -lpthread -lm -lrt -lutil")
endif()
configure_file(cmake/libunshield.pc.in libunshield.pc @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libunshield.pc" DESTINATION "${CMAKE_INSTALL_LIBDIR}/pkgconfig")
configure_package_config_file(cmake/unshieldConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/unshieldConfig.cmake"
    INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/unshield"
    PATH_VARS CMAKE_INSTALL_LIBDIR CMAKE_INSTALL_INCLUDEDIR CMAKE_INSTALL_BINDIR)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/unshieldConfigVersion.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/unshieldConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/unshieldConfigVersion.cmake"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/unshield")
if(BUILD_TESTING)
    if(TEST_REAL_INSTALLERS)
        find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
        add_executable(real_members tests/reference_members.c)
        target_link_libraries(real_members PRIVATE libunshield)
        target_compile_features(real_members PRIVATE c_std_11)
        if(MSVC)
            target_compile_definitions(real_members PRIVATE _CRT_SECURE_NO_WARNINGS)
            target_include_directories(real_members PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
        endif()
        if(WIN32 AND NOT BUILD_STATIC)
            add_custom_command(TARGET real_members POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RUST_SHARED}" "$<TARGET_FILE_DIR:real_members>")
        endif()
        foreach(IO_MODE native callbacks)
            add_test(NAME real_installers_${IO_MODE}
                COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/real-installers.py"
                    check-c --extractor $<TARGET_FILE:real_members> --io ${IO_MODE})
        endforeach()
    endif()
    add_executable(c_abi tests/c_abi.c)
    target_link_libraries(c_abi PRIVATE libunshield)
    target_compile_features(c_abi PRIVATE c_std_11)
    # assert is the test runner, so preserve it for Release builds too.
    if(MSVC)
        target_compile_options(c_abi PRIVATE /UNDEBUG)
        target_compile_definitions(c_abi PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_include_directories(c_abi PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
    else()
        target_compile_options(c_abi PRIVATE -UNDEBUG)
    endif()
    if(WIN32 AND NOT BUILD_STATIC)
        add_custom_command(TARGET c_abi POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RUST_SHARED}" "$<TARGET_FILE_DIR:c_abi>")
    endif()
    add_test(NAME generate_c_fixtures
        COMMAND "${CMAKE_COMMAND}" -E env ${RUST_ENV} "${CARGO_EXECUTABLE}" run --locked --example make-c-fixtures -- "${CMAKE_CURRENT_BINARY_DIR}/c-fixtures")
    set_tests_properties(generate_c_fixtures PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" FIXTURES_SETUP c_fixtures)
    add_test(NAME c_abi COMMAND c_abi "${CMAKE_CURRENT_BINARY_DIR}/c-fixtures")
    set_tests_properties(c_abi PROPERTIES FIXTURES_REQUIRED c_fixtures)
endif()
if(TEST_UPSTREAM_CLI)
    set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/upstream/unshield")
    if(NOT EXISTS "${UPSTREAM}/src/unshield.c")
        message(FATAL_ERROR "Fetch the pinned upstream revision using scripts/fetch-upstream first")
    endif()
    # Upstream includes errno.h only with iconv enabled, although its glibc
    # extraction path also uses errno. Supply it without modifying the source.
    add_executable(unshield_c tests/upstream_cli.c)
    target_link_libraries(unshield_c PRIVATE libunshield)
    target_compile_definitions(unshield_c PRIVATE SIZE_FORMAT="zi")
    if(MSVC)
        target_include_directories(unshield_c PRIVATE "${UPSTREAM}/win32_msvc")
        target_compile_definitions(unshield_c PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    if(WIN32 AND NOT BUILD_STATIC)
        add_custom_command(TARGET unshield_c POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${RUST_SHARED}" "$<TARGET_FILE_DIR:unshield_c>")
    endif()
    if(BUILD_TESTING)
        add_test(NAME original_c_cli COMMAND unshield_c t "${CMAKE_CURRENT_BINARY_DIR}/c-fixtures/v6/data1.cab")
        set_tests_properties(original_c_cli PROPERTIES FIXTURES_REQUIRED c_fixtures)
    endif()
endif()
