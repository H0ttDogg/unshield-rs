cmake_minimum_required(VERSION 3.21)
project(unshield_consumer LANGUAGES C)
find_package(unshield 1 CONFIG REQUIRED)
add_executable(consumer ../c_abi.c)
target_link_libraries(consumer PRIVATE unshield::libunshield)
target_compile_features(consumer PRIVATE c_std_11)
if(MSVC)
    target_compile_options(consumer PRIVATE /UNDEBUG)
    target_compile_definitions(consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_include_directories(consumer PRIVATE ..)
else()
    target_compile_options(consumer PRIVATE -UNDEBUG)
endif()
get_target_property(UNSHIELD_LIBRARY_TYPE unshield::libunshield TYPE)
if(WIN32 AND UNSHIELD_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:unshield::libunshield> $<TARGET_FILE_DIR:consumer>)
endif()
