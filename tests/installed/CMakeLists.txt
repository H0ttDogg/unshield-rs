cmake_minimum_required(VERSION 3.21)
project(unshield_consumer LANGUAGES C)
include(CTest)
option(TEST_REAL_INSTALLERS "Require real installers through the installed library" OFF)
find_package(unshield 1 CONFIG REQUIRED)
get_target_property(UNSHIELD_LIBRARY_TYPE unshield::libunshield TYPE)
add_executable(consumer ../c_abi.c)
target_link_libraries(consumer PRIVATE unshield::libunshield)
target_compile_features(consumer PRIVATE c_std_11)
if(MSVC)
    target_compile_options(consumer PRIVATE /UNDEBUG)
    target_compile_definitions(consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_include_directories(consumer PRIVATE ..)
else()
    target_compile_options(consumer PRIVATE -UNDEBUG)
endif()
if(TEST_REAL_INSTALLERS)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    add_executable(real_consumer ../reference_members.c)
    target_link_libraries(real_consumer PRIVATE unshield::libunshield)
    target_compile_features(real_consumer PRIVATE c_std_11)
    if(MSVC)
        target_compile_definitions(real_consumer PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_include_directories(real_consumer PRIVATE ..)
    endif()
    if(WIN32 AND UNSHIELD_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
        add_custom_command(TARGET real_consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:unshield::libunshield> $<TARGET_FILE_DIR:real_consumer>)
    endif()
    foreach(IO_MODE native callbacks)
        add_test(NAME installed_real_installers_${IO_MODE}
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/real-installers.py"
                check-c --extractor $<TARGET_FILE:real_consumer> --io ${IO_MODE})
    endforeach()
endif()
if(WIN32 AND UNSHIELD_LIBRARY_TYPE STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET consumer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:unshield::libunshield> $<TARGET_FILE_DIR:consumer>)
endif()
